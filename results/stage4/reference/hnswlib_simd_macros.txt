hnswlib enables its hand-written SIMD distance kernels only when compiling for x86 (SSE/AVX); on arm64 (Apple M2) it uses its generic scalar C++ loops.

#ifndef NO_MANUAL_VECTORIZATION
#if (defined(__SSE__) || _M_IX86_FP > 0 || defined(_M_AMD64) || defined(_M_X64))
#define USE_SSE
#ifdef __AVX__
#define USE_AVX
#ifdef __AVX512F__
#define USE_AVX512
#endif
#endif
#endif
#endif
